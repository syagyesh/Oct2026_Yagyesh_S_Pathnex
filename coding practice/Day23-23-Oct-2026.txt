- name: Deploy AWS Lambda Function
  hosts: localhost
  tasks:
    - name: Create Lambda function
      command: aws lambda create-function --function-name yagyesh-lambda --runtime python3.8 --role arn:aws:iam::123456789012:role/lambda-execution-role --handler lambda_function.lambda_handler --zip-file fileb://lambda.zip

resource "aws_api_gateway_rest_api" "yagyesh_api" {
  name        = "yagyesh-api"
  description = "yagyesh API"
}

resource "aws_lambda_function" "yagyesh_lambda" {
  function_name = "yagyesh-lambda"
  runtime       = "python3.8"
  role          = aws_iam_role.lambda_exec_role.arn
  handler       = "lambda_function.lambda_handler"
  filename      = "lambda.zip"
}

resource "aws_api_gateway_integration" "lambda_integration" {
  rest_api_id = aws_api_gateway_rest_api.yagyesh_api.id
  resource_id = aws_api_gateway_resource.yagyesh_resource.id
  http_method = aws_api_gateway_method.yagyesh_method.http_method
  integration_http_method = "POST"
  type = "AWS_PROXY"
  uri  = aws_lambda_function.yagyesh_lambda.invoke_arn
}

helm install yagyesh-microservice ./microservice-chart


pipeline {
    agent any
    stages {
        stage('Build') {
            steps {
                echo 'Building Lambda function...'
                sh 'zip -r lambda.zip lambda/'
            }
        }
        stage('Deploy to AWS Lambda') {
            steps {
                script {
                    sh 'aws lambda update-function-code --function-name yagyesh-lambda --zip-file fileb://lambda.zip'
                }
            }
        }
    }
}

# Multi-Stage Docker Build
# Stage 1 — Build Stage
FROM maven:3.9-eclipse-temurin-17 AS builder
WORKDIR /opt/yagyesh/java-build
COPY . /opt/yagyesh/java-build/
RUN mvn clean package

# Stage 2 — Production Stage
FROM tomcat:9
WORKDIR /usr/local/tomcat/webapps
COPY --from=builder /opt/yagyesh/java-build/target/*.war /usr/local/tomcat/webapps/
EXPOSE 8080
CMD ["catalina.sh", "run"]
