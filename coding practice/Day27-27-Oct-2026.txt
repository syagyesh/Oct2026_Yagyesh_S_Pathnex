- name: Deploy Serverless Lambda Function
  hosts: localhost
  tasks:
    - name: Create Lambda function
      command: aws lambda create-function --function-name yagyesh-lambda --runtime nodejs14.x --role arn:aws:iam::123456789012:role/lambda-role --handler index.handler --zip-file fileb://lambda.zip
resource "aws_dynamodb_table" "yagyesh_table" {
  name           = "yagyesh-table"
  hash_key       = "id"
  billing_mode   = "PAY_PER_REQUEST"
  attribute {
    name = "id"
    type = "S"
  }
}
apiVersion: kubeless.io/v1beta1
kind: Function
metadata:
  name: yagyesh-lambda
  namespace: default
spec:
  runtime: nodejs14
  handler: index.handler
  function: |
    module.exports = function(event, context) {
      context.succeed('Hello World');
    };
pipeline {
    agent any
    stages {
        stage('Build') {
            steps {
                echo 'Building Lambda function package...'
                sh 'zip -r lambda.zip lambda/'
            }
        }
        stage('Deploy to Lambda') {
            steps {
                script {
                    sh 'aws lambda update-function-code --function-name yagyesh-lambda --zip-file fileb://lambda.zip'
                }
            }
        }
    }
}
